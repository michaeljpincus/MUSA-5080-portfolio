The cross-validation helper

Code 
ctrl <- trainControl(method = "cv", number = 10)

cv_rmse <- function(formula, data) {
  df <- if (inherits(data, "sf")) st_drop_geometry(data) else data
  vars <- all.vars(formula)
  bad <- vars[sapply(vars, function(v) any(is.na(df[[v]])))]
  if (length(bad) > 0) {
    stop("These variables have NAs: ", paste(bad, collapse = ", "),
         ". Fill them or choose another variable (Rule 1: don't drop rows).")
  }
  set.seed(5080)
  fit <- train(formula, data = df, method = "lm", trControl = ctrl)
  fit$results$RMSE
}

philly_tracts <- philly_tracts %>%
  mutate(
    # Dependent variable: % with no vehicle
    y     = no_vehE / total_hhE * 100,
    y_moe = tidycensus::moe_prop(no_vehE, total_hhE, no_vehM, total_hhM) * 100,
    y_cv  = (y_moe / 1.645) / y * 100,
    
    # Predictors
    pct_rentersE = rentersE / total_occE * 100,
    pct_rentersM = tidycensus::moe_prop(rentersE, total_occE, rentersM, total_occM) * 100,
    
    tt60plusE = tt60E + tt90E,
    tt60plusM = sqrt(tt60M^2 + tt90M^2),
    pct60ttE  = tt60plusE / tt_workE * 100,
    pct60ttM  = tidycensus::moe_prop(tt60plusE, tt_workE, tt60plusM, tt_workM) * 100
  )

#| label: cv-helper
ctrl <- trainControl(method = "cv", number = 10)

cv_rmse <- function(formula, data) {
  df <- if (inherits(data, "sf")) st_drop_geometry(data) else data
  vars <- all.vars(formula)
  bad <- vars[sapply(vars, function(v) any(is.na(df[[v]])))]
  if (length(bad) > 0) {
    stop("These variables have NAs: ", paste(bad, collapse = ", "),
         ". Fill them or choose another variable (Rule 1: don't drop rows).")
  }
  set.seed(5080)
  fit <- train(formula, data = df, method = "lm", trControl = ctrl)
  fit$results$RMSE
}
